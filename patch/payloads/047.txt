function nameOf(name) {

function descriptor(file) {

export function preparationRequest(key) {
  check(typeof key === 'string' && key.length < 1024, 'Invalid PDF preparation key.');
  let parsed; try { parsed = JSON.parse(key); } catch { check(false); }
  const [version, pdfjs, sha256, settings] = parsed || [];
  check(Array.isArray(parsed) && parsed.length === 4 && Number.isSafeInteger(version) && version > 0
    && /^\d+\.\d+\.\d+$/.test(pdfjs) && /^[a-f0-9]{64}$/.test(sha256)
    && settings && ['auto', 'single', 'double'].includes(settings.mode)
    && ['keepMarginNotes', 'keepRepeatedText', 'normalizeColors', 'reflow'].every(k => typeof settings[k] === 'boolean')
    && [20, 24, 28].includes(settings.textSize) && Object.keys(settings).length === 6,
  'Invalid PDF preparation settings.');
  return { version, pdfjs, sha256, settings };
}

// Keep only checked identities in memory. Reuse the checksum while the file is
// unchanged, and read each requested range without loading/parsing the whole
// book again for every chunk. Original-file and small-metadata limits remain.
const preparedFilecountPreparedWords(layout) {
  if (!Array.isArray(layout?.blocks) || layout.preparation?.complete === false) return null;
  let words = 0;
  for (const block of layout.blocks) {
    if (!block.reflow || block.annotation || block.annotationLabel) continue;
    let text = '', previous;
    // Same source-word reconstruction as the reader, including words which
    // wrap across rows. OCR metadata behind an image is not readable text.
    for (const row of block.reflow.rows) {
      if (text && (!previous || row.label || previous.label || row.start !== previous.end)) text += ' ';
      text += row.text; previous = row;
    }
    words += (text.match(/\S*[\p{L}\p{N}]\S*/gu) || []).length;
  }
  return words;
}
