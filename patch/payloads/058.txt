function backend() {
    const id = randomUUID(), cloudId = 'cse_nativeTest', calls = [], events = [];
    let binding = { localId: id, cloudId, pid: 123, procStart: '456', cwd: '/test' };
    let metadata = { id: cloudId, title: 'Native test', status: 'active', environment_kind: 'bridge', connection_status: 'connected',
        worker_status: 'idle', requires_action_details_list: [], created_at: '2026-09-24T00:00:00Z' };
    const append = payload => events.push({ sequence_num: String(events.length + 1), event_id: randomUUID(), payload: { uuid: randomUUID(), session_id: cloudId, ...payload } });
    const message = (role, text, uuid = randomUUID()) => append({ type: role, uuid, parent_tool_use_id: null, message: { role, content: text } });
    message('user', 'Desktop message'); message('assistant', 'Desktop reply');
    const client = {
        async listSessions() { calls.push(['list']); return [structuredClone(metadata)]; },
        async getSession(remote) { assert.equal(remote, cloudId); calls.push(['metadata']); return structuredClone(metadata); },
        async events(remote, { cursor, order = 'asc', limit = 200 } = {}) {
            assert.equal(remote, cloudId); calls.push(['events', cursor, order]);
            const filtered = events.filter(e => !cursor || (order === 'desc' ? BigInt(e.sequence_num) < BigInt(cursor) : BigInt(e.sequence_num) > BigInt(cursor)));
            if (order === 'desc') filtered.reverse();
            const data = filtered.slice(0, limit);
            return structuredClone({ data, next_cursor: filtered.length > data.length ? data.at(-1).sequence_num : null });
        },
        async send(remote, payload) {
            assert.equal(remote, cloudId); calls.push(['send', structuredClone(payload)]);
            append(payload);
            if (payload.type === 'user') {
                append(payload); // Some transports echo the same user UUID.('assistant', 'Reply to glasses');
            } else if (payload.type === 'control_response') {
                metadata.worker_status = 'idle'; metadata.requires_action_details_list = []results: [{}] };
        },
        close() { calls.push(['close']); },
    };
    return { id, cloudId, calls, events, append, message, client, bindings: () => binding ? [binding] : [],
        switchSession: () => { binding = { ...binding, localId: randomUUID() }; },
        setMetadata: patch => { metadata = { ...metadata, ...patch }; } };
}

test('native client uses existing OAuth session/event endpoints and never retries an ambiguous send', async t => {
    const dir = await mkdtemp(join(tmpdir(), 'even-native-client-'));
    t.after(() => rm(dir, { recursive: true, force: true }));
    await writeFile(join(dir, '.credentials.json'), JSON.stringify({ claudeAiOauth: { accessToken: 'test-only-token', expiresAt: Date.now() + 60000 } }));
    const calls = [];
    const client = new ClaudeRemoteClient({ configDir: dir, fetchImpl: async (url, options) => {
        calls.push([url, options]);
        assert.equal(options.headers.Authorization, 'Bearer test-only-token');
        if (options.method === 'POST') throw new Error('Connection lost after sending');
        return new Response(JSON.stringify({ response_shape: { id: 'cse_test', environment_kind: 'bridge' } }));
    } });
    assert.equal((await client.getSession('cse_test')).id, 'cse_test');
    const payload = { uuid: randomUUID(), type: 'user', session_id: 'cse_test', message: { role: 'user', content: 'Test' } };
    await assert.rejects(client.send('cse_test', payload), /delivery is unknown/);
    assert.equal(calls.length, 2);
    assert.equal(calls[1][0], 'https://api.anthropic.com/v1/code/sessions/cse_test/events');
    assert.deepEqual(JSON.parse(calls[1][1].body), { events: [{ payload }] });
    client.close();
    await assert.rejects(client.getSession('cse_test'), /closed/);
    assert.equal(calls.length, 2);
});

test('native client reports credential rejection without exposing the token', async t => {
    const dir = await mkdtemp(join(tmpdir(), 'even-native-auth-'));
    t.after(() => rm(dir, { recursive: true, force: true }));
    await writeFile(join(dir, '.credentials.json'), JSON.stringify({ claudeAiOauth: { accessToken: 'private-token', expiresAt: Date.now() + 60000 } }));
    let requests = 0;
    const client = new ClaudeRemoteClient({ configDir: dir, fetchImpl: async () => {
        requests++; return new Response(JSON.stringify({ error: { message: 'Rejected private-token' } }), { status: 403 });
    } });
    t.after(() => client.close());
    await assert.rejects(client.listSessions(), error => error.statusCode === 403 && !error.message.includes('private-token'));
    assert.equal(requests, 1);
});
