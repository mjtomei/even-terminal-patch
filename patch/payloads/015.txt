export async function runChannel() {
    const channel = new ClaudeChannel(process.env.EVEN_CLAUDE_LAUNCH_DIR);
    const mcp = new Server({ name: 'even', version: '0.1.0' }, {
        capabilities: { experimental: { 'claude/channel': {}, 'claude/channel/permission': {} }, tools: {} },
        instructions: 'Messages from the even channel are the user speaking from their Even glasses or phone in this same conversation. Answer these messages using the reply tool. Copy prompt_id, session_id and generation from the channel metadata. The reply is shown on the glasses and in this terminal; do not repeat it as a second ordinary assistant message. For terminal and native Remote Control prompts, respond normally. Keep your existing instructions and permission settings.',
    });
    channel.notify = message => mcp.notification(message);
    mcp.setRequestHandler(ListToolsRequestSchema, () => ({ tools: [{ name: 'reply', description: 'Reply to a message from the user’s Even glasses or phone.',
        inputSchema: { type: 'object', properties: { prompt_id: { type: 'string' }, session_id: { type: 'string' }, generation: { type: 'string' }, text: { type: 'string' } },
            required: ['prompt_id', 'session_id', 'generation', 'text'], additionalProperties: false } }] }));
    mcp.setRequestHandler(CallToolRequestSchema, async request => {
        if (request.params.name !== 'reply') throw new Error('Unknown Even tool');
        try { return await channel.reply(request.params.arguments || {}); }
        catch (error) { return { isError: true, content: [{ type: 'text', text: error.message }] }; }
    });
    mcp.setNotificationHandler(z.object({ method: z.literal('notifications/claude/channel/permission_request'), params: z.object({
        request_id: z.string(), tool_name: z.string(), description: z.string(), input_preview: z.string(),
    }) }), notification => channel.permission(notification.params));
    mcp.oninitialized = () => { channel.ready = true; channel.register(); };
    mcp.onclose = () => { void channel.close().finally(() => process.exit(0)); };
    await channel.start();
    await mcp.connect(new StdioServerTransport());
    const shutdown = async () => { await channel.close(); await mcp.close(); process.exit(0); };
    process.on('SIGINT', shutdown); process.on('SIGTERM', shutdown);
}
