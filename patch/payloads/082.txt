    throw new Error('Condition did not become true');
}

test('stale history cannot erase phone/glasses turns after sending, reconnecting or restarting', async t => {

    const completed = [];
    for (const [n, prompt, answer] of [[2, 'Phone test', 'Phone received'], [3, 'Hello?', 'Glasses received']]) {
        const turn = { id: `turn-${n}`, status: 'inProgress', startedAt: d.thread.turns[0].startedAt + 1, items: [] };

        turn.items.push(user, agent);

        d.emit('item/agentMessage/delta', { threadId: 'existing', turnId: turn.id, itemId: agent.id, delta: answer });
        d.emit('item/completed', { threadId: 'existing', turnId: turn.id, item: agent });
        d.complete();
        completed.push(prompt, answer);
        await until(() => events.some(m => m.type === 'result' && m.text === answer));
    }
    // Both history APIs still show the old desktop turn, as on the real daemon.

    await provider.prompt('existing', 'Next glasses input');

    assert.equal(events.slice(eventCount).some(m => m.type === 'result' || m.text === 'Desktop question'), false);

    d.complete();
    await until(() => provider.getStatus('existing').state === 'idle');
    d.thread.turns = structuredClone(d.persistedTurns);
    client.ws.terminate();
    await until(() => d.requests.filter(r => r.method === 'thread/resume').length >= 2 && provider.getStatus('existing').connected);

    await provider.close();
    const reopened = providerFor(t, d, { stateDir }).provider;
    assert.deepEqual((await reopened.getHistory('existing', 10)).slice(-4).map(m => m.text), completed);
    assert.deepEqual((await reopened.getConversation('existing')).filter(t => ['turn-2', 'turn-3'].includes(t.id)).flatMap(t => t.items.map(i => i.text)), completed);

    process.env.CODEX_HOME = home;
    t.after(async () => {
        if (previousHome === undefined) delete process.env.CODEX_HOME;
        else process.env.CODEX_HOME = previousHome;
        await rm(home, { recursive: true, force: true });
    });
    await mkdir(join(home, 'sessions'))home, 'sessions', 'large.jsonl');
    const row = (turn, text) => JSON.stringify({ type: 'event_msg', timestamp: '2026-10-14T12:00:00Z',
        payload: { type: 'item_completed', turn_id: turn, item: { id: turn + '-user', type: 'UserMessage', content: [{ type: 'text', text }] } } });
    const header = JSON.stringify({ type: 'session_meta', payload: { id: 'existing' } });
    const tool = JSON.stringify({ type: 'response_item', payload: { output: 'x'.repeat(9 * 1024 * 1024) } });
    await writeFile(path, [header, row('old', 'Old message outside recovery window'), tool, '\0bad-record', row('recent', 'Recent café 🥽'), ''].join('\n'));
    const session = { id: 'existing', transcript: new CodexTranscript() };

    assert.deepEqual(remote.transcript.history(), []);
});

});
