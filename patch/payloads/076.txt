import test from 'node:test';
import assert from 'node:assert/strict';
import {mkdtemp,writeFile,mkdir,rm} from 'node:fs/promises';
import {tmpdir} from 'node:os';
import path from 'node:path';
import {createHash} from 'node:crypto';
import {libraryManifest,libraryPreparation,libraryOcrDirectory} from '../dist/routes/pageborne-library.js';
const hash=value=>createHash('sha256').update(value).digest('hex');
test('optional OCR serves only complete matching preparations and leaves originals alone',async t=>{
 const libraryDir=await mkdtemp(path.join(tmpdir(),'terminal-ocr-'));t.after(()=>rm(libraryDir,{recursive:true,force:true}));
 const options={libraryDir},name='scan.pdf',engine='a'.repeat(64);
 await writeFile(path.join(libraryDir,name),'%PDF-original');
 const manifest=await libraryManifest({},options);
 const key=JSON.stringify([17,'6.3.289',manifest.files[0].sha256,{mode:'auto',keepMarginNotes:false,keepRepeatedText:false,normalizeColors:true,reflow:true,textSize:20}]);
 assert.equal((await libraryPreparation({name,key},options)).state,'unavailable');
 const root=await libraryOcrDirectory(options);
 await mkdir(path.join(root,engine,'prepared'),{recursive:true});await mkdir(path.join(root,'requests'));
 await writeFile(path.join(root,'daemon.json'),JSON.stringify({engine,preparationVersion:17,pdfjs:'6.3.289',heartbeat:Date.now()}));
 assert.equal((await libraryPreparation({name,key},options)).state,'pending');
 const artifact=JSON.stringify({key,engine,saved:{version:17,layout:{}}});
 await writeFile(path.join(root,engine,'prepared',hash(key)+'.json'),artifact);
 const ready=await libraryPreparation({name,key},options);assert.equal(ready.state,'ready');
 const chunk=await libraryPreparation({name,key,offset:0,sha256:ready.sha256},options);
 assert.equal(Buffer.from(chunk.data,'base64').toString(),artifact);
 await assert.rejects(libraryPreparation({name,key,offset:0,sha256:'b'.repeat(64)},options));
 await writeFile(path.join(libraryDir,name),'%PDF-changed');
 await assert.rejects(libraryPreparation({name,key},options),/changed/i);
});
