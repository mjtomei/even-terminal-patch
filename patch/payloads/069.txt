function handler(router, path, method = 'get') {
    return router.stack.find(layer => layer.route?.path === path && layer.route.methods[method]).route.stack[0].handle;
}
function response() {
    return { statusCode: 200, headers: {}, frames: [], destroyed: false,
        status(code) { this.statusCode = code; return this; },
        json(body) { this.body = body; return this; },
        setHeader(name, value) { this.headers[name] = value; },
        flushHeaders() {},
        write(frame) { this.frames.push(frame); return true; } };
}

test('listed Claude sessions open through status and SSE without starting a query', async t => {
    t.mock.method(ClaudeSession.prototype, 'start', () => assert.fail('Opening a saved session started Claude'));
    t.mock.method(ClaudeSession.prototype, 'run', () => assert.fail('Opening a saved session sent a prompt'));
    const f = await savedSession([['user', 'From the desktop'], ['assistant', [{ type: 'text', text: 'Saved reply' }]]]);
    const before = await readFile(f.path, 'utf8'createClaudeProvider(() => assert.fail('Opening a saved session emitted a new message'));
    assert.ok((50)).some(session => session.id === f.id));
    const req = Object.assign(new EventEmitter(), { query: { provider: 'claude', sessionId: f.id }, headers: {} });
    const status = response();
    await handler(core, '/status')(req, status);
    assert.equal(status.statusCode, 200);
    assert.deepEqual(status.body, { state: 'idle', sessionId: f.id, provider: 'claude' });
    const stream = response();
    try {
        await handler(events, '/events')(req, stream);
        assert.equal(stream.headers['Content-Type'], 'text/event-stream');
        const data = stream.frames.flatMap(frame => frame.split('\n')).filter(line => line.startsWith('data: ')).map(line => JSON.parse(line.slice(6)));
        assert.ok(data.some(message => message.type === 'status' && message.state === 'idle' && message.sessionId === f.id));
    } finally { req.emit('close'); }
    assert.equal(await readFile(f.path, 'utf8'), before);
});

test('glasses can open and address a listed Claude session without a provider field', async t => {
    t.mock.method(codexProvider, 'attach', () => assert.fail('Claude ID was sent to Codex'));
    const f = await savedSession([['user', 'Desktop prompt'], ['assistant', 'Saved reply']]);
    const beforeListing = response();
    await handler(core, '/status')({ query: { sessionId: f.id } }, beforeListing);
    assert.equal(beforeListing.body.provider, 'claude');
    const listed = response();
    await handler(core, '/sessions')({ query: { provider: 'claude', limit: '100' } }, listed);
    assert.ok(listed.body.sessions.some(session => session.id === f.id));
    const req = Object.assign(new EventEmitter(), { query: { sessionId: f.id }, headers: {} });
    const stream = response();
    try {
        await handler(events, '/events')(req, stream);
        assert.ok(stream.frames.some(frame => frame.includes('"provider":"claude"')));
    } finally { req.emit('close'); }
    for (const path of ['/status', '/messages']) {
        const res = response();
        await handler(core, path)(req, res);
        assert.equal(res.body.provider, 'claude');
    }
    const history = response();
    await handler(core, '/sessions/:id/history')({ params: { id: f.id }, query: {} }, history);
    assert.deepEqual(history.body, { history: [{ role: 'user', text: 'Desktop prompt' }, { role: 'assistant', text: 'Saved reply' }] });
    const calls = [];
    t.mock.method(claudeProvider, 'prompt', async (id, text) => { calls.push(['prompt', id, text]); return 'claude' }; });
    for (const method of ['respondPermission', 'respondQuestion', 'interrupt']) {
        t.mock.method(claudeProvider, method, (...args) => calls.push([method, ...args]));
    }
    const prompt = response();
    await handler(core, '/prompt', 'post')({ body: { sessionId: f.id, text: 'Glasses prompt' } }, prompt);
    assert.equal(prompt.statusCode, 202);
    assert.equal(prompt.body.provider, 'claude');
    for (const path of ['/permission-response', '/question-response', '/interrupt']) {
        const res = response();
        await handler(core, path, 'post')({ body: { sessionId: f.id, decision: 'allow', answer: 'yes', toolUseId: 'request' } }, res);
        assert.equal(res.statusCode, 200);
    }
    assert.deepEqual(calls, [['prompt', f.id, 'Glasses prompt'], ['respondPermission', f.id, 'allow', 'request'],
        ['respondQuestion', f.id, 'yes', 'request'], ['interrupt', f.id]]);
});
