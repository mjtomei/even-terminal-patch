// Extra display detail for the authenticated Pageborne conversation endpoint.
// Legacy tool summaries and events continue to use the existing summarizers.
export function commandDetail(label, command) {
    if (typeof command !== 'string' || !command.trim()) return undefined;
    const text = `${label} ${command.replace(/\s+/g, ' ').trim()}`<= 1000 ? text : text.slice(0, 999).replace(/[\uD800-\uDBFF]$/, '') + '…';
}
