export function defaultEndpoint(env = process.env) {
    return env.CODEX_APP_SERVER_URL || `unix://${env.CODEX_APP_SERVER_SOCKET || join(env.CODEX_HOME || join(homedir(), ".codex"), "app-server-control", "app-server-control.sock")}`;
}

export function connectionOptions(endpoint, env = process.env{ perMessageDeflate: false, handshakeTimeout: 5000, maxPayload: 64 * 1024 * 1024 };
    if (endpoint.startsWith("unix://")) {
        const path = endpoint.slice(7);
        if (!isAbsolute(path) || path.includes(":")) throw new Error("Codex socket must be an absolute path without ':'");
        return { url: `ws+unix://${path}:/`, options: { ...options, headers: { Host: "localhost" } } };
    }
    const url = new URL(endpoint);
    if (!["ws:", "wss:"].includes(url.protocol)) throw new Error("Codex endpoint must use unix://, ws://, or wss://");

    if (token) options.headers = { Authorization: `Bearer ${token}` };
    return { url: endpoint, options };
}
