import test from 'node:test';
import assert from 'node:assert/strict';
import { mkdtemp, rm, readFile, readdir, symlink, mkdir } from 'node:fs/promises';
import os from 'node:os';
import path from 'node:path';
import { paperWorkspace, expandPaperBase } from '../dist/routes/pageborne-workspaces.js';

test('paper folders use a hidden home default, remain stable across titles, and isolate identities and custom bases', async t => {
  const home = await mkdtemp(path.join(os.tmpdir(), 'pageborne-workspaces-')); t.after(() => rm(home, { recursive: true, force: true }));
  const first = await paperWorkspace({ workspaceId: 'session-one', paperId: 'one', title: 'Paper' }, { home });
  assert.equal(first.baseDir, path.join(home, '.pageborne/papers'));
  const same = await paperWorkspace({ workspaceId: 'session-one', paperId: 'one', title: 'Renamed title' }, { home }); assert.equal(same.cwd, first.cwd);
  const anotherSession = await paperWorkspace({ paperId: 'one', workspaceId: 'session-two' }, { home }); assert.notEqual(anotherSession.cwd, first.cwd); assert.equal(path.dirname(anotherSession.cwd), path.dirname(first.cwd));
  const other = await paperWorkspace({ workspaceId: 'session-one', paperId: 'two', title: 'Paper' }, { home }); assert.notEqual(other.cwd, first.cwd);
  const custom = await paperWorkspace({ workspaceId: 'session-one', paperId: 'one', baseDir: '~/research' }, { home }); assert.equal(custom.baseDir, path.join(home, 'research'));
  assert.equal(JSON.parse(await readFile(path.join(first.cwd, 'paper.json'))).title, 'Paper');
  const hostile = await paperWorkspace({ workspaceId: 'session-one', paperId: '../../elsewhere', title: '../../secret' }, { home }); assert.equal(path.dirname(path.dirname(hostile.cwd)), first.baseDir);
  assert.throws(() => expandPaperBase('relative/path', home)); assert.throws(() => expandPaperBase('~somebody/papers', home));
  await rm(first.cwd, { recursive: true }); const elsewhere = path.join(home, 'elsewhere'); await mkdir(elsewhere); await symlink(elsewhere, first.cwd);
  await assert.rejects(paperWorkspace({ workspaceId: 'session-one', paperId: 'one' }, { home }), /symbolic link/); assert.deepEqual(await readdir(elsewhere), []);
});
