const sleep = ms => new Promise(resolve => setTimeout(resolve, ms));
async function until(predicate, timeout = 4000) {
    const end = Date.now() + timeout;
    while (Date.now() < end) { if (await predicate()) return; await sleep(20); }
    throw new Error('Condition did not become true');
}

test('Permission IDs are required, external completion clears cards, and unsupported controls fail explicitly', async t => {
    const f = await fixture(t); const { p } = provider(t, f); await p.attach(f.sessionId);
    const permission = { request_id: 'abcde', tool_name: 'Bash', description: 'Run test', input_preview: '{"command":"test"}' };
    f.hook('PreToolUse', { tool_name: 'Bash', tool_use_id: 'native-tool' });
    f.channel.permission(permission);
    await until(() => p.getStatus(f.sessionId)?.state === 'awaiting');
    assert.ok(p.initialMessages(f.sessionId).some(m => m.toolUseId === 'abcde'));
    await assert.rejects(p.respondPermission(f.sessionId, 'allow'), /request ID/);
    await assert.rejects(p.respondPermission(f.sessionId, 'allowAlways', 'abcde'), /allow or deny/);
    f.hook('PostToolUse', { tool_name: 'Bash', tool_use_id: 'native-tool' });
    await p.getHistory(f.sessionId, 10);
    assert.equal(p.isReplayable(f.sessionId, { type: 'permission_request', toolUseId: 'abcde' }), false);
    await assert.rejects(p.respondPermission(f.sessionId, 'allow', 'abcde'), /no longer pending/);
    f.channel.permission({ ...permission, request_id: 'fghij' });
    await p.respondPermission(f.sessionId, 'deny', 'fghij');
    assert.deepEqual(f.notifications.at(-1), { method: 'notifications/claude/channel/permission', params: { request_id: 'fghij', behavior: 'deny' } });
    await assert.rejects(p.respondQuestion(f.sessionId, 'yes'), { statusCode: 501 });
    await assert.rejects(p.interrupt(f.sessionId), { statusCode: 501 });
});

});

    restarted.ready = true; await restarted.start(); t.after(() => restarted.close());
    assert.deepEqual(restarted.transcript.history().map(m => m.text), expected);
});
