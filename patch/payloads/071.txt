import test from 'node:test';
import assert from 'node:assert/strict';
import { mkdtempSync, rmSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { CodexTranscript } from '../dist/codex/transcript.js';
import { RemoteTranscript } from '../dist/claude/remote/transcript.js';
import { conversationTurns } from '../dist/routes/pageborne-conversation.js';

test('Codex activities retain display summaries through streaming and restart without contaminating answers', () => {
    const dir = mkdtempSync(join(tmpdir(), 'activity-')transcript = new CodexTranscript('fixture', dir);
        transcript.merge([{ id: 'turn', status: 'inProgress', items: [{ id: 'u', type: 'userMessage', content: [{ type: 'text', text: 'Question' }] }] }]);
        const notify = (method, item) => transcript.notification(method, { turnId: 'turn', ...item });
        notify('item/started', { item: { id: 'think', type: 'reasoning', content: ['SECRET RAW'], encryptedContent: 'SECRET ENCRYPTED' } });
        notify('item/reasoning/summaryTextDelta', { itemId: 'think', delta: 'Checking the relevant file.' });
        notify('item/completed', { item: { id: 'think', type: 'reasoning' } });
        notify('item/started', { item: { id: 'cmd', type: 'commandExecution', command: 'cat todo.txt', aggregatedOutput: 'SECRET OUTPUT' } });
        let turn = conversationTurns(transcript.snapshot())[0];
        assert.equal(turn.answer, ''); assert.equal(turn.activities[0].text, 'Checking the relevant file.');
        assert.equal(turn.activities[0].state, 'complete'); assert.equal(turn.activities[1].state, 'pending');
        notify('item/completed', { item: { id: 'cmd', type: 'commandExecution', command: 'cat todo.txt', status: 'failed' } });
        notify('item/completed', { item: { id: 'a', type: 'agentMessage', text: 'Answer' } });
        transcript.flush();
        const restored = new CodexTranscript('fixture', dir);
        turn = conversationTurns(restored.snapshot())[0];
        assert.equal(turn.activities[1].state, 'error'); assert.equal(turn.answer, 'Answer');
        assert.doesNotMatch(JSON.stringify(restored.snapshot()), /SECRET/);
        assert.deepEqual(restored.history().map(i => i.text), ['Question', 'Answer']);
        restored.merge([{ id: 'turn', items: [{ id: 'cmd', type: 'commandExecution', command: 'cat todo.txt', status: 'inProgress' }] }]);
        assert.equal(conversationTurns(restored.snapshot())[0].activities[1].state, 'error');
        restored.flush();
    } finally { rmSync(dir, { recursive: true, force: true }); }
});

test('Claude stores exposed thinking and tool lifecycle, excluding tool results and redacted blocks', () => {
    const transcript = new RemoteTranscript('fixture', () => {});
    let n = 0;
    const ingest = (type, content, extra = {}) => transcript.ingest({ sequence_num: ++n, payload: { type, message: { content }, ...extra } }, false);
    ingest('user', 'Question');
    ingest('assistant', [{ type: 'thinking', thinking: 'Checking relevant files.', signature: 'SECRET SIGNATURE' },
        { type: 'redacted_thinking', data: 'SECRET REDACTED' }, { type: 'tool_use', id: 'read', name: 'Read', input: { file_path: '/project/agents.md' } }]);
    let turns = conversationTurns(transcript.snapshot('busy'));
    assert.equal(turns.length, 1); assert.equal(turns[0].answer, ''); assert.equal(turns[0].activities.length, 2);
    assert.equal(turns[0].activities[1].state, 'pending');
    ingest('user', [{ type: 'tool_result', tool_use_id: 'read', content: 'SECRET FILE' }]);
    ingest('assistant', [{ type: 'text', text: 'Answer' }]); ingest('result', [], { is_error: false });
    turns = conversationTurns(transcript.snapshot());
    assert.equal(turns.length, 1); assert.equal(turns[0].activities[1].state, 'complete'); assert.equal(turns[0].answer, 'Answer');
    assert.doesNotMatch(JSON.stringify(transcript.snapshot()), /SECRET/);
    assert.deepEqual(transcript.history().map(i => i.text), ['Question', 'Answer']);
    assert.deepEqual(transcript.initialMessages().map(i => i.text.trim()), ['Question', 'Answer']);
});
