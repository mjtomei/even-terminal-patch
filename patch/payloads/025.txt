import { open, realpath } from "node:fs/promisejoin, sep } from "node:path";

const MAX_ROLLOUT_BYTES = 8 * 1024 * 1024;
const MAX_HEADER_BYTES = 1024 * 1024;

async function readRange(file, offset, length) {
    const buffer = Buffer.alloc(length);
    let read = 0;
    while (read < length) {
        const { bytesRead } = await file.read(buffer, read, length - read, offset + read);
        if (!bytesRead) break;
        read += bytesRead;
    }
    return buffer.subarray(0, read);
}

async function recentRollout(file, size) {
    if (size <= MAX_ROLLOUT_BYTES) return (await readRange(file, 0, size)).toString("utf8");
    // Recover recent messages without loading a multi-hundred-MB rollout.
    // Keep the first record so session identity is still checked. Reading one
    // byte before the tail preserves complete records at an exact boundary.
    const header = await readRange(file, 0, Math.min(size, MAX_HEADER_BYTES));
    const headerEnd = header.indexOf(10);
    if (headerEnd < 0) return "";
    const metadata = JSON.parse(header.subarray(0, headerEnd).toString("utf8"));
    if (metadata.type !== "session_meta") return "";
    const start = size - MAX_ROLLOUT_BYTES - 1;
    const tail = await readRange(file, start, MAX_ROLLOUT_BYTES + 1);
    const firstNewline = tail.indexOf(10);
    if (firstNewline < 0) return "";
    return header.subarray(0, headerEnd).toString("utf8") + "\n" + tail.subarray(firstNewline + 1).toString("utf8");
}
