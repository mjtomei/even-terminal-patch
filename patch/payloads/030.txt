const toolTypes = new Set(['commandExecution', 'fileChange', 'mcpToolCall', 'dynamicToolCall',
    'webSearch', 'imageView', 'imageGeneration', 'collabAgentToolCall']);
function displayItem(item) {
    if (item.type !== 'reasoning' && !toolTypes.has(item.type)) return item;
    // Keep display summaries and bounded shell command detail. Never expose
    // reasoning content, encrypted payloads, arbitrary tool input/output or signatures.
    const summary = Array.isArray(item.summary) ? item.summary.map(part => typeof part === 'string' ? part : part?.text || '').join('\n') : '';
    const detail = item.type === 'commandExecution' ? commandDetail('Shell', item.command) : undefined;
    return { id: item.id, type: 'activity', kind: item.type === 'reasoning' ? 'thinking' : 'tool',
        text: item.type === 'reasoning' ? (summary || 'Thinking…').slice(0, 800) .slice(0, 200),
        ...(detail ? { detail } : {}),
        state: /failed|error|declined/.test(item.status || '') ? 'error'
            : item.complete || /completed|done/.test(item.status || '') ? 'complete' : 'pending',
        complete: item.complete };
}
