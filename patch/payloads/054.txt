import QRCode from "qrcode-terminal/vendor/QRCode/index.js";
import levels from "qrcode-terminal/vendor/QRCode/QRErrorCorrectLevel.js";

export function pairingSvg(url) {
    const qr = new QRCode(-1, levels.M);
    qr.addData(url);
    qr.make();
    const size = qr.getModuleCount();
    const cells = [];
    for (let row = 0; row < size; row++) {
        for (let col = 0; col < size; col++) {
            if (qr.isDark(row, col)) cells.push(`M${col + 4},${row + 4}h1v1h-1zreturn `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${size + 8} ${size + 8}" width="640" height="640" shape-rendering="crispEdges"><title>Scan with the Even app to connect</title><rect width="100%" height="100%" fill="white"/><path d="${cells.join("")}" fill="black"/></svg>\n`;
}
