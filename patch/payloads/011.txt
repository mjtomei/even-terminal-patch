import path from 'node:path';

import { fail } from './client.js';

const execute = promisify(execFile);
const quote = value => `'${String(value).replaceAll("'", "'\\''")}'`;
const uuid = value => typeof value === 'string' && /^[0-9a-f]{8}(?:-[0-9a-f]{4}){3}-[0-9a-f]{12}$/i.test(value);

// A normal interactive CLI owns each conversation, independently of the HTTP
// bridge. A private tmux server keeps it available to native Remote Control.
// Creation is explicitly enabled on this desktop; existing sessions are never
// resumed, forked, or given a different model or permission configuration.
export function createNativeSessionLauncher({ model, cli = process.env.PAGEBORNE_CLAUDE_CLI || 'claude',
    stateDir = path.join(homedir(), '.pageborne', 'claude-sessions'),
    defaultCwd = path.join(homedir(), '.pageborne', 'library'),
    run = execute, bindings = localBindings, timeoutMs = 20000 } = {}) {
    if (!model) return undefined;
    const tasks = new Map();
    async function prepare({ requestId, cwd = defaultCwd, title = 'Pageborne' }) {
        if (!uuid(requestId) || typeof cwd !== 'string' || !path.isAbsolute(cwd) || /[\x00-\x1f]/.test(cwd) ||
            typeof title !== 'string' || [...title].length > 300 || /[\x00-\x1f]/.test(title)) throw fail('Invalid Claude session request.', 400);
        cwd = await realpath(cwd);
        if (!(await stat(cwd)).isDirectory()) throw fail('Claude session directory is unavailable.', 400);
        title = title.trim() || 'Pageborne';
        await mkdir(stateDir, { recursive: true, mode: 0o700 });
        const record = { sessionId: requestId, cwd, title, model };
        const file = path.join(stateDir, `${requestId}.json`);
        let fresh = false;
        try { await writeFile(file, JSON.stringify(record), { flag: 'wx', mode: 0o600 }); fresh = true; }
        catch (error) {
            if (error.code !== 'EEXIST') throw error;
            const saved = JSON.parse(await readFile(file, 'utf8'));
            if (saved.cwd !== cwd || saved.title !== title || saved.model !== model) throw fail('This session request already belongs to another conversation.', 409tching = () => bindings().filter(b => b.localId === requestId && b.cwd === cwd);
        if (matching().length === 1) return record;
        const name = `pb-${requestId}`;
        const exists = async () => { try { await run('tmux', ['-L', 'pageborne-claude', 'has-session', '-t', name], { timeout: 5000 }); return true; } catch { return false; } };
        if (!await exists()) {
            if (!fresh) throw fail('The shared Claude session has closed. Start a new session in Ask, or reopen it on the desktop.', 409);
            const args = [cli, '--model', model, '--permission-mode', 'default', '--remote-control', title, '--name', title, '--session-id', requestId];
            try { await run('tmux', ['-L', 'pageborne-claude', 'new-session', '-d', '-s', name, '-c', cwd, `exec ${args.map(quote).join(' ')}`], { timeout: 10000 }); }
            catch { throw fail('Could not start the shared Claude desktop session. Check the Claude CLI and tmux installation.', 503);imeoutMs;
              if (matching().length === 1) return record;
            await new Promise(resolve => setTimeout(resolve, 200row fail('Claude is waiting for desktop setup. Open its Pageborne session on the desktop, then retry; your question has not been sent.', 503);
    }
    return { model, prepare(input) {
        // Serialise identical retries, including during connection loss. The
        // durable reservation prevents another launch after a bridge restart.
        const key = JSON.stringify(input);
        if (!tasks.has(key)) tasks.set(key, prepare(input).finally(() => tasks.delete(key)));
        return tasks.get(key);
    } };
}
